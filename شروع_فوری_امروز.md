```python
# ایجاد فایل retry_utils.py در ریشه پروژه:
import time
import logging
from functools import wraps

logger = logging.getLogger(__name__)

def retry_with_backoff(max_retries=3, backoff_factor=2, base_wait=1):
    """Decorator for retry with exponential backoff."""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    last_exception = e
                    if attempt == max_retries - 1:
                        logger.error(f"Max retries reached for {func.__name__}: {e}")
                        raise
                    
                    wait_time = base_wait * (backoff_factor ** attempt)
                    logger.warning(f"Retry {attempt + 1}/{max_retries} after {wait_time}s")
                    time.sleep(wait_time)
            return wrapper
        return wrapper
    return decorator

# تست decorator
def test_retry():
    call_count = 0
    
    @retry_with_backoff(max_retries=3)
    def failing_function():
        nonlocal call_count
        call_count += 1
        if call_count < 3:
            raise ConnectionError("Simulated failure")
        return "Success"
    
    result = failing_function()
    print(f"Result: {result}, Calls: {call_count}")
    return result == "Success" and call_count == 3

if __name__ == "__main__":
    test_retry()
```

```python
import time
import pandas as pd
import numpy as np
from trading_ai_system.data.data import DataFetcher
from trading_ai_system.features.features import FeatureEngineer

def measure_performance():
    """Measure current system performance."""
    print("📊 Performance Baseline Measurement")
    print("=" * 50)
    
    # 1. Data loading performance
    print("\n1. Data Loading Performance:")
    start = time.time()
    
    # Create sample data (since we might not have real data)
    dates = pd.date_range('2023-01-01', periods=1000, freq='D')
    sample_data = pd.DataFrame({
        'open': np.random.randn(1000).cumsum() + 100,
        'high': np.random.randn(1000).cumsum() + 102,
        'low': np.random.randn(1000).cumsum() + 98,
        'close': np.random.randn(1000).cumsum() + 101,
        'volume': np.random.randint(1000, 10000, 1000)
    }, index=dates)
    
    data_time = time.time() - start
    print(f"   Sample data creation: {data_time:.3f}s")
    
    # 2. Feature calculation performance
    print("\n2. Feature Calculation Performance:")
    start = time.time()
    
    engineer = FeatureEngineer()
    features = engineer.calculate(sample_data)
    
    feature_time = time.time() - start
    print(f"   Feature calculation: {feature_time:.3f}s")
    print(f"   Features shape: {features.shape}")
    
    # 3. Memory usage
    print("\n3. Memory Usage:")
    import psutil
    process = psutil.Process()
    memory_mb = process.memory_info().rss / 1024 / 1024
    print(f"   Current memory: {memory_mb:.1f} MB")
    
    # Summary
    print("\n📈 Performance Summary:")
    print(f"   Total time: {data_time + feature_time:.3f}s")
    print(f"   Memory: {memory_mb:.1f} MB")

        'data_time': data_time,
        'feature_time': feature_time,
        'total_time': data_time + feature_time,
        'memory_mb': memory_mb,
        'timestamp': time.time()
    }
    
    import json
    with open('performance_baseline.json', 'w') as f:
        json.dump(baseline, f)
    
    print(f"\n✅ Baseline saved to performance_baseline.json")
    return baseline

if __name__ == "__main__":
    measure_performance()
```
